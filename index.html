repository/MemoryMemberRepository.java HashMap으로 구현
package com.example.hello.repository;

import com.example.hello.domain.Member;
import org.springframework.stereotype.Repository;
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicLong;

@Repository   // Spring이 이 객체를 만들어 관리(빈 등록)
public class MemoryMemberRepository implements MemberRepository {

    private final Map<Long, Member> store = new ConcurrentHashMap<>();
    private final AtomicLong sequence = new AtomicLong();

    @Override
    public Member save(Member member) {
        member.setId(sequence.incrementAndGet());
        store.put(member.getId(), member);
        return member;
    }

    @Override
    public Optional<Member> findById(Long id) {
        return Optional.ofNullable(store.get(id));
    }

    @Override
    public Optional<Member> findByName(String name) {
        return store.values().stream()
                .filter(m -> m.getName().equals(name))
                .findFirst();
    }

    @Override
    public List<Member> findAll() {
        return new ArrayList<>(store.values());
    }
}
service/MemberService.java 규칙은 여기에만
package com.example.hello.service;

import com.example.hello.domain.Member;
import com.example.hello.repository.MemberRepository;
import org.springframework.stereotype.Service;
import java.util.List;

@Service
public class MemberService {

    private final MemberRepository memberRepository;

    // 생성자가 하나면 Spring이 알아서 MemoryMemberRepository를 넣어줌 (DI)
    public MemberService(MemberRepository memberRepository) {
        this.memberRepository = memberRepository;
    }

    public Long join(String name) {
        memberRepository.findByName(name).ifPresent(m -> {
            throw new IllegalStateException("이미 존재하는 회원입니다.");
        });
        return memberRepository.save(new Member(name)).getId();
    }

    public Member findOne(Long id) {
        return memberRepository.findById(id)
                .orElseThrow(() -> new IllegalArgumentException("회원이 없습니다. id=" + id));
    }

    public List<Member> findAll() {
        return memberRepository.findAll();
    }
}

controller/MemberController.java URL과 Service를 연결
package com.example.hello.controller;

import com.example.hello.dto.MemberCreateRequest;
import com.example.hello.dto.MemberResponse;
import com.example.hello.service.MemberService;
import org.springframework.web.bind.annotation.*;
import java.util.List;

@RestController                  // 반환값을 JSON으로 응답
@RequestMapping("/api/members")   // 이 클래스의 공통 URL
public class MemberController {

    private final MemberService memberService;

    public MemberController(MemberService memberService) {
        this.memberService = memberService;
    }

    @PostMapping                     // POST /api/members
    public MemberResponse create(@RequestBody MemberCreateRequest request) {
        Long id = memberService.join(request.name());
        return MemberResponse.from(memberService.findOne(id));
    }

    @GetMapping                      // GET /api/members
    public List<MemberResponse> list() {
        return memberService.findAll().stream()
                .map(MemberResponse::from)
                .toList();
    }

    @GetMapping("/{id}")            // GET /api/members/1
    public MemberResponse get(@PathVariable Long id) {
        return MemberResponse.from(memberService.findOne(id));
    }
}
exception/GlobalExceptionHandler.java 예외를 500 대신 400 + 메시지로
package com.example.hello.exception;

import org.springframework.http.HttpStatus;
import org.springframework.web.bind.annotation.*;
import java.util.Map;

@RestControllerAdvice
public class GlobalExceptionHandler {

    @ResponseStatus(HttpStatus.BAD_REQUEST)
    @ExceptionHandler({IllegalStateException.class, IllegalArgumentException.class})
    public Map<String, String> handle(RuntimeException e) {
        return Map.of("message", e.getMessage());
    }
}
